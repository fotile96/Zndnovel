```python
"""提案预聚类：把车道 notes/*.md 的散条提案按词族聚合成裁决卷宗（P3 spec §2/§4）。
段末小裁决会话的输入整理器——聚合跨批/跨章的同词提案，并对每个词族附
saturation 饱和度判定（≥gate 可裁决，否则滚存）。只读，不改任何 notes 文件。"""
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
from tools.pipelib import read_jsonl
from tools.resolve import resolve_map
from tools.saturation import saturation

BATCH_RE = re.compile(r"^## batch (\d+)\s*$")
PROP_RE = re.compile(r"^### (\d+)\.\s*(.+)$")
# 标题里包裹词项的装饰：markdown 粗体、引号、括号注（含 (7302) 这类 id 注）
STRIP_RE = re.compile(r"\*\*|[『』「」]|[（(][^）)]*[）)]")
UNKEYED = "（未归词）"


def extract_terms(title: str) -> list:
    t = re.sub(r"^\d+\.\s*", "", title)
    if "→" not in t:
        return []
    left = STRIP_RE.sub("", t.split("→", 1)[0]).replace("★", "")
    return [w.strip() for w in left.split("/") if w.strip()]


def parse_notes(text: str, chapter: str) -> list:
    props, batch, cur = [], None, None
    for line in text.splitlines():
        if m := BATCH_RE.match(line):
            batch = int(m.group(1))
            continue
        if m := PROP_RE.match(line):
            if cur:
                props.append(cur)
            title = m.group(2)
            cur = {"chapter": chapter, "batch": batch, "num": int(m.group(1)),
                   "title": title, "terms": extract_terms(title),
                   "starred": "★" in title, "body": ""}
            continue
        if cur is not None and not line.startswith("## "):
            cur["body"] += line + "\n"
        elif cur is not None:
            props.append(cur)
            cur = None
    if cur:
        props.append(cur)
    return props


def cluster(proposals: list) -> dict:
    c = {}
    for p in proposals:
        for key in (p["terms"] or [UNKEYED]):
            c.setdefault(key, []).append(p)
    return c


def main(argv=None):
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--lane-dir", required=True)
    ap.add_argument("--gate", type=float, default=0.8)
    ap.add_argument("--out", help="省略则打印 stdout")
    a = ap.parse_args(argv)
    lane = Path(a.lane_dir)
    data = lane / "data"
    props = []
    for p in sorted((data / "notes").glob("*.md")):
        props += parse_notes(p.read_text(encoding="utf-8"), p.stem)
    ledger = read_jsonl(data / "ledger.jsonl")
    extra = data / "ledger_extra.jsonl"
    if extra.exists():
        ledger += read_jsonl(extra)
    fwd = read_jsonl(data / "forward.jsonl") if (data / "forward.jsonl").exists() else []
    pat = read_jsonl(data / "patches.jsonl") if (data / "patches.jsonl").exists() else []
    done = set(resolve_map(fwd, pat))
    c = cluster(props)
    lines = [f"# 术语提案裁决卷宗（{lane.name} 车道，共 {len(props)} 条提案 / {len(c)} 个词族）",
             ""]
    def _order(item):
        # 多提案词族在前（跨批冲突优先裁），未归词殿后
        key, ps = item
        return (key == UNKEYED, -len(ps), key)
    for key, ps in sorted(c.items(), key=_order):
        if key == UNKEYED:
            head = f"## {UNKEYED}（{len(ps)} 条，逐条人工过目）"
        else:
            s = saturation(key, ledger, done)
            verdict = ("无出现" if s["ratio"] is None
                       else "可裁决" if s["ratio"] >= a.gate else "滚存")
            ratio = "-" if s["ratio"] is None else f"{s['ratio']:.3f}"
            head = (f"## {key}（{len(ps)} 条提案 | 饱和度 {s['covered']}/{s['total']}"
                    f"={ratio} → **{verdict}**）")
        lines.append(head)
        for p in ps:
            lines.append(f"### [{p['chapter']} b{p['batch']} #{p['num']}] {p['title']}")
            lines.append(p["body"].rstrip())
            lines.append("")
    text = "\n".join(lines)
    if a.out:
        Path(a.out).write_text(text, encoding="utf-8")
        print(f"卷宗已写 {a.out}（{len(props)} 提案 / {len(c)} 词族）")
    else:
        print(text)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

# F9：作品内引用的日文原文（钱仙拼字、五十音教学等）在译文中合法保留假名。
# 判据：假名连串长度 ≤ 本值且逐字出现在同行 jp 原文中 → 视为引用、不计入假名比。
# 长串即使出现在 jp 中也不豁免——整行未译（zh≈jp）正是 C1 的目标。
JP_QUOTE_MAX_LEN = 6
KANA_RUN_RE = re.compile(r"[぀-ゟァ-ヺヽ-ヿ]+")


def _kana_ratio(zh: str, names, jp: str = "") -> float:
    s = zh
    for n in sorted(names, key=len, reverse=True):
        s = s.replace(n, "")
    if jp:
        s = KANA_RUN_RE.sub(
            lambda m: "" if len(m.group()) <= JP_QUOTE_MAX_LEN and m.group() in jp
            else m.group(), s)
    return len(KANA_RE.findall(s)) / max(len(zh), 1)
```

"""外部 API 薄客户端（P3 spec §5 里程碑 1）：deepseek 首连，OpenAI 兼容 chat/completions。
纯 stdlib（urllib）；key 取环境变量 DEEPSEEK_API_KEY 或仓库根 .env（已 gitignore，
绝不入库）；每次调用向 data/api_ledger.jsonl 追加一行计费台账（token 计数为准，
换算金额留给人看账时按当日价目做）。transport 可注入以便离线测试。
提供商无关的扩展点是 PROVIDERS 表——加第二家只需加一条记录。"""
import json
import os
import sys
import time
import urllib.request
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
from tools.pipelib import REPO

PROVIDERS = {
    "deepseek": {"url": "https://api.deepseek.com/chat/completions",
                 "key_env": "DEEPSEEK_API_KEY",
                 "default_model": "deepseek-chat"},
}


def load_env(path=None) -> dict:
    p = Path(path) if path else REPO / ".env"
    env = {}
    if p.exists():
        for line in p.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if line and not line.startswith("#") and "=" in line:
                k, v = line.split("=", 1)
                env[k.strip()] = v.strip().strip('"').strip("'")
    return env


def _http_transport(url, headers, payload):
    req = urllib.request.Request(url, data=json.dumps(payload).encode("utf-8"),
                                 headers=headers, method="POST")
    with urllib.request.urlopen(req, timeout=300) as resp:
        return json.loads(resp.read().decode("utf-8"))


def chat(messages, *, provider="deepseek", model=None, temperature=0.2,
         max_tokens=2048, purpose="", transport=None, ledger_path=None,
         api_key=None) -> dict:
    cfg = PROVIDERS[provider]
    model = model or cfg["default_model"]
    key = api_key or os.environ.get(cfg["key_env"]) or load_env().get(cfg["key_env"])
    if not key:
        raise SystemExit(
            f"缺 {cfg['key_env']}：设环境变量，或写进仓库根 .env（已 gitignore）")
    payload = {"model": model, "messages": messages,
               "temperature": temperature, "max_tokens": max_tokens}
    headers = {"Content-Type": "application/json", "Authorization": f"Bearer {key}"}
    resp = (transport or _http_transport)(cfg["url"], headers, payload)
    usage = resp.get("usage", {})
    out = {"content": resp["choices"][0]["message"]["content"],
           "prompt_tokens": usage.get("prompt_tokens", -1),
           "completion_tokens": usage.get("completion_tokens", -1),
           "model": model, "provider": provider}
    lp = Path(ledger_path) if ledger_path else REPO / "data" / "api_ledger.jsonl"
    with open(lp, "a", encoding="utf-8") as f:
        f.write(json.dumps(
            {"ts": time.strftime("%Y-%m-%dT%H:%M:%S%z"), "provider": provider,
             "model": model, "purpose": purpose,
             "prompt_tokens": out["prompt_tokens"],
             "completion_tokens": out["completion_tokens"]},
            ensure_ascii=False) + "\n")
    return out


def main(argv=None):
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--prompt", required=True)
    ap.add_argument("--provider", default="deepseek")
    ap.add_argument("--model")
    ap.add_argument("--purpose", default="cli-smoke")
    a = ap.parse_args(argv)
    r = chat([{"role": "user", "content": a.prompt}], provider=a.provider,
             model=a.model, purpose=a.purpose)
    print(r["content"])
    print(f"[{r['provider']}/{r['model']} tokens {r['prompt_tokens']}+{r['completion_tokens']}]",
          file=sys.stderr)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

注意：`chat` 缺 key 是 `SystemExit`（不是静默降级）；`test_missing_key_exits_with_guidance` 用 monkeypatch 挡住真实 `.env`——本机可能已有 key，测试不得依赖也不得读它。

```python
"""彩野测试（P3 spec §5）：让从未见过原文的外部模型扮演「只靠梗概续听故事的读者」，
只喂 压缩前缀（skeleton/cards.zh.yaml 卡组梗概＝游戏自身的『あらすじ』机制）＋
本段中文译文（零日文原文），回答剧情理解题；答错处＝译文或压缩丢了信息，且丢的
恰是真实玩家会丢的。判分由第二次 API 调用完成。advisory 工具：恒 exit 0，
结论供段末人工验收参考。--dry-run 只组装统计、零 API 调用。"""
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
import yaml

from tools.pipelib import REPO, read_jsonl
from tools.resolve import resolve_map
from tools.api_client import chat

READER_SYS = ("你在听人讲一个故事。你只知道两部分信息：①此前剧情的卡片梗概；"
              "②最新一段正文。只根据给定文本回答问题，文本里没有的就答「不知道」，"
              "不要编造，不要使用任何文本之外的知识。用中文简答，两三句以内。")
JUDGE_SYS = ("你是判分员。根据【问题】【标准答案】【受试回答】判定：对／部分对／错。"
             "只输出一行，格式：判定|一句话依据。")


def assemble_context(label: str, lane_dir: Path, repo: Path = REPO):
    segs = yaml.safe_load((repo / "config" / "segments.yaml").read_text(encoding="utf-8"))
    hit = next((s for s in segs if s["label"] == label), None)
    if hit is None:
        raise SystemExit(f"未知段 label：{label}")
    lo, hi = hit["batches"]
    data = Path(lane_dir) / "data"
    batches = {r["batch"]: r for r in read_jsonl(data / "batches.jsonl")}
    fwd = read_jsonl(data / "forward.jsonl") if (data / "forward.jsonl").exists() else []
    pat = read_jsonl(data / "patches.jsonl") if (data / "patches.jsonl").exists() else []
    zh = resolve_map(fwd, pat)
    lines, missing = [], 0
    for n in range(lo, hi + 1):
        if n not in batches:
            continue
        for i in batches[n]["ids"]:
            if i in zh:
                lines.append(zh[i])
            else:
                missing += 1
    cards = yaml.safe_load((repo / "skeleton" / "cards.zh.yaml").read_text(encoding="utf-8"))
    prefix = "\n\n".join(f"【卡{c['card']}·{c.get('title_zh', '')}】\n{c.get('body_zh', '')}"
                         for c in cards)
    ctx = ("＝＝ 此前剧情的卡片梗概 ＝＝\n\n" + prefix
           + "\n\n＝＝ 最新一段正文 ＝＝\n\n" + "\n".join(lines))
    return ctx, missing


def main(argv=None, *, transport=None, api_key=None, ledger_path=None):
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--segment", required=True)
    ap.add_argument("--qa", required=True)
    ap.add_argument("--lane-dir", required=True)
    ap.add_argument("--repo", default=None, help="仅测试用：覆盖仓库根")
    ap.add_argument("--out", default=None)
    ap.add_argument("--dry-run", action="store_true")
    a = ap.parse_args(argv)
    repo = Path(a.repo) if a.repo else REPO
    ctx, missing = assemble_context(a.segment, Path(a.lane_dir), repo=repo)
    qa = yaml.safe_load(Path(a.qa).read_text(encoding="utf-8"))
    if a.dry_run:
        print(f"segment={a.segment} 上下文 {len(ctx)} 字符 缺译 {missing} 行 题数 {len(qa)}")
        return 0
    kw = dict(transport=transport, api_key=api_key, ledger_path=ledger_path)
    rows = []
    for q in qa:
        ans = chat([{"role": "system", "content": READER_SYS},
                    {"role": "user", "content": f"{ctx}\n\n＝＝ 问题 ＝＝\n{q['question']}"}],
                   purpose=f"ayano:{a.segment}:{q['id']}", **kw)
        verdict = chat([{"role": "system", "content": JUDGE_SYS},
                        {"role": "user", "content":
                         f"【问题】{q['question']}\n【标准答案】{q['reference']}\n"
                         f"【受试回答】{ans['content']}"}],
                       purpose=f"ayano-judge:{a.segment}:{q['id']}", **kw)
        rows.append((q, ans["content"], verdict["content"]))
    out = Path(a.out) if a.out else repo / "work" / "out" / f"ayano-{a.segment}.md"
    out.parent.mkdir(parents=True, exist_ok=True)
    lines = [f"# 彩野测试报告 · {a.segment}（{len(rows)} 题）", ""]
    if missing:
        lines.append(f"> ⚠ 本段**缺译 {missing} 行**（段未译完，结论仅供参考）。")
        lines.append("")
    for q, ans, verdict in rows:
        lines += [f"## {q['id']}：{q['question']}",
                  f"- 信息点：{q.get('info_point', '-')}",
                  f"- 标准答案：{q['reference']}",
                  f"- 受试回答：{ans}",
                  f"- **判分：{verdict}**", ""]
    out.write_text("\n".join(lines), encoding="utf-8")
    print(f"报告已写 {out}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
